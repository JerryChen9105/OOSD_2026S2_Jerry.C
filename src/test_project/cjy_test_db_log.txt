create table course (
courseID int,
courseName varchar(60) not null,
requiredTotalCredit float not null,

constraint pk_courseID primary key (courseID)
);

create table study_field (
fieldPrefix char(4),
fieldName varchar(60),

constraint pk_fieldPrefix primary key (fieldPrefix)
);

create table subject (
fieldPrefix char(4),
subjectCode int,
subjectName varchar(60),
creditPoint float not null,

constraint pk_subjectID primary key (fieldPrefix, subjectCode),
constraint fk_subjectPrefix foreign key (fieldPrefix)
	references study_field(fieldPrefix)
);

create table major (
majorID int,
fieldPrefix char(4) not null,
majorName varchar(60),
requiredCredit float not null,

constraint pk_majorID primary key (majorID),
constraint fk_majorField foreign key (fieldPrefix)
	references study_field(fieldPrefix)
);

create table student (
sid int,
fname varchar(40),
lname varchar(40),
courseID int,
birthday datetime not null,
majorID int,

constraint pk_sid primary key (sid),
constraint fk_courseID foreign key (courseID)
	references course(courseID)
    on delete set null,
constraint fk_majorID foreign key (majorID)
	references major(majorID)
    on delete set null
);

create table student_enrolment (
sid int,
fieldPrefix char(4),
subjectCode int,
constraint pk_enrolmentID primary key (sid, fieldPrefix, subjectCode),
constraint fk_sid foreign key (sid)
	references student(sid)
    on update cascade,
constraint fk_subjectID foreign key (fieldPrefix, subjectCode)
	references subject(fieldPrefix, subjectCode)
    on update cascade
);
